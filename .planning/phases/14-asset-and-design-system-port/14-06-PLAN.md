Recommended GRD skeleton (top-level GRD calling into the grdp):

  <grit latest_public_release="0" current_release="1">
    <outputs>
      <output filename="prometheus_resources.pak" type="data_package" />
      <output filename="grit/prometheus_resources.h" type="rc_header">
        <emit emit_type="prepend"></emit>
      </output>
    </outputs>
    <release seq="1">
      <includes>
        <include name="IDR_PROMETHEUS_FONT_POPPINS_THIN_WOFF2"
                 file="fonts/Poppins-Thin.woff2" type="BINDATA" />
        <!-- ... all font/icon/mark entries ... -->
      </includes>
    </release>
  </grit>

GN grit() target pattern (from chrome/browser/resources BUILD.gn convention):

      "grit/prometheus_resources.h",

    ]
  }

  PrometheusThemePreviewUI(content::WebUI* web_ui) : content::WebUIController(web_ui) {
    auto* profile = Profile::FromWebUI(web_ui);
    auto* source = content::WebUIDataSource::CreateAndAdd(profile, "prometheus-theme-preview");
    source->SetDefaultResource(IDR_PROMETHEUS_THEME_PREVIEW_HTML);
    // Map EVERY asset the HTML references so nothing renders blank:
    source->AddResourcePath("prometheus_fonts.css", IDR_PROMETHEUS_CSS_FONTS);
    // brand marks:
    source->AddResourcePath("images/pm_mark_dark.png", IDR_PROMETHEUS_ASSET_PM_MARK_DARK); // ...and light/wordmark/fsb
    // 69 FA svgs — add each, e.g.:
    source->AddResourcePath("svg/fa_bolt.svg", IDR_PROMETHEUS_ICON_FA_BOLT);               // ...all 69
    // fonts (woff2) referenced by prometheus_fonts.css @font-face url(//resources/prometheus/fonts/...):
    source->AddResourcePath("fonts/Poppins-Thin.woff2", IDR_PROMETHEUS_FONT_POPPINS_THIN_WOFF2); // ...all 5
    // Prefer driving these AddResourcePath calls from a generated webui::ResourcePath array (the GRD's
    // grit/prometheus_resources_map.h kPrometheusResources[]) so all manifest entries are added in one loop
    // rather than 80 hand-written lines. Alternatively serve shared fonts/icons from chrome://resources/.
    webui::SetupWebUIDataSource(source, base::span(kPrometheusResources), IDR_PROMETHEUS_THEME_PREVIEW_HTML);
  }
